# Blog CMS Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Move blog posts from `.md` files into a `blog_posts` table and add a browser editor at `/admin/blog`, so posts are written and published without touching files or redeploying.

**Architecture:** New `blog_posts` table. `lib/blog.ts` rewritten to read the DB (published only) instead of the filesystem; the `<Markdown>` render path is unchanged. Admin CRUD lives in `lib/admin-blog.ts` + `app/admin/blog/*`, gated by the existing `requireAdmin()`. Pages stay fast via build-time prerender + on-demand `revalidatePath` on save.

**Tech Stack:** Next.js App Router (server components + one client editor), Drizzle ORM (Postgres), react-markdown + remark-gfm (already installed), gray-matter (already installed, import only), Vitest.

## Global Constraints

- Deployed on Vercel — runtime filesystem is read-only, so posts must live in the DB, not files written at runtime.
- Public pages (`/blog`, `/blog/[slug]`) show only `status = 'published'`; admin sees all statuses. (This changes today's behavior, which lists drafts publicly with a badge.)
- Existing 5 `.md` files in `apps/web/content/blog/` are imported once (idempotent, keyed on slug); the DB becomes the source of truth. Files stay in the repo as history.
- Rendering is unchanged: `<Markdown>{markdownString}</Markdown>` (`components/blog/markdown.tsx`).
- Reuse `estimateReadingMinutes` (private in `lib/blog.ts`) — do not reinvent.
- Code comments: one line, ≤20 words.
- Migrations: `pnpm --filter web db:generate` generates; Rachit runs `pnpm --filter web db:migrate`.
- Final gate: `pnpm --filter web build` + `pnpm --filter web test` + `pnpm --filter web lint`.
- Test harness: `import { db, resetDb, schema, seedUser } from "./helpers"`, vitest. Pure-helper tests must `import "./env-bootstrap"` first (lib files import `@/lib/db` at load).

**Reference (verified):**
- `BlogFrontmatter` and `BlogPost` types + `getAllPosts`/`getPost` live in `lib/blog.ts`.
- Frontmatter fields on disk: `title, description, slug, targetKeyword, secondaryKeywords (list), intent, status, order`.
- Admin action pattern: `"use server"`, `await requireAdmin()`, mutate, `revalidatePath(...)` (`app/admin/actions.ts`).
- Timestamps helper `...timestamps` (createdAt/updatedAt) in `lib/db/schema.ts`.
- Drizzle quotes identifiers, so a column named `order` (a SQL reserved word) is safe.

```ts
import "server-only"

import { asc, desc, eq, sql } from "drizzle-orm"

import { db, schema } from "@/lib/db"

export type BlogFrontmatter = {
  title: string
  description: string
  slug: string
  targetKeyword?: string
  secondaryKeywords?: string[]
  intent?: "informational" | "transactional" | "commercial"
  status?: "draft" | "published"
  order?: number
}

export type BlogPost = {
  slug: string
  frontmatter: BlogFrontmatter
  content: string
  readingMinutes: number
}

export type BlogPostRow = typeof schema.blogPosts.$inferSelect

export function estimateReadingMinutes(text: string): number {
  const words = text.trim().split(/\s+/).filter(Boolean).length
  return Math.max(1, Math.round(words / 220))
}

// Pure row → BlogPost mapping so pages and the editor preview agree on shape.
export function rowToBlogPost(row: BlogPostRow): BlogPost {
  return {
    slug: row.slug,
    frontmatter: {
      title: row.title,
      description: row.description,
      slug: row.slug,
      targetKeyword: row.targetKeyword ?? undefined,
      secondaryKeywords: (row.secondaryKeywords as string[]) ?? [],
      intent: row.intent ?? undefined,
      status: row.status,
      order: row.order ?? undefined,
    },
    content: row.content,
    readingMinutes: estimateReadingMinutes(row.content),
  }
}

// Published posts for the public site, ordered by `order` (nulls last) then
// newest published first.
export async function getAllPosts(): Promise<BlogPost[]> {
  const rows = await db
    .select()
    .from(schema.blogPosts)
    .where(eq(schema.blogPosts.status, "published"))
    .orderBy(sql`${schema.blogPosts.order} asc nulls last`, desc(schema.blogPosts.publishedAt))
  return rows.map(rowToBlogPost)
}

export async function getPost(slug: string): Promise<BlogPost | null> {
  const row = await db.query.blogPosts.findFirst({
    where: (p, { and }) => and(eq(p.slug, slug), eq(p.status, "published")),
  })
  return row ? rowToBlogPost(row) : null
}
```

Note: `asc` is imported for readability but the order clause uses raw `sql` for `nulls last`; if lint flags `asc` as unused, drop it from the import.

```ts
import { and, desc, eq, ne } from "drizzle-orm"

import { db, schema } from "@/lib/db"
import type { BlogPostRow } from "@/lib/blog"

export interface AdminBlogRow {
  id: string
  slug: string
  title: string
  status: "draft" | "published"
  order: number | null
  updatedAt: Date
}

export interface SavePostInput {
  id?: string
  slug: string
  title: string
  description: string
  targetKeyword?: string
  secondaryKeywords: string[]
  intent?: "informational" | "transactional" | "commercial"
  status: "draft" | "published"
  order?: number
  content: string
}

export async function listPostsAdmin(): Promise<AdminBlogRow[]> {
  return db
    .select({
      id: schema.blogPosts.id,
      slug: schema.blogPosts.slug,
      title: schema.blogPosts.title,
      status: schema.blogPosts.status,
      order: schema.blogPosts.order,
      updatedAt: schema.blogPosts.updatedAt,
    })
    .from(schema.blogPosts)
    .orderBy(desc(schema.blogPosts.updatedAt))
}

export async function getPostAdmin(id: string): Promise<BlogPostRow | null> {
  const row = await db.query.blogPosts.findFirst({ where: eq(schema.blogPosts.id, id) })
  return row ?? null
}

// Upsert. publishedAt is stamped the first time status becomes published and
// never cleared, so unpublish→republish keeps the original date.
export async function savePost(input: SavePostInput): Promise<{ id: string }> {
  // Slug must be unique across other posts.
  const clash = await db.query.blogPosts.findFirst({
    where: input.id
      ? and(eq(schema.blogPosts.slug, input.slug), ne(schema.blogPosts.id, input.id))
      : eq(schema.blogPosts.slug, input.slug),
    columns: { id: true },
  })
  if (clash) throw new Error("Slug already in use")

  const fields = {
    slug: input.slug,
    title: input.title,
    description: input.description,
    targetKeyword: input.targetKeyword ?? null,
    secondaryKeywords: input.secondaryKeywords,
    intent: input.intent ?? null,
    status: input.status,
    order: input.order ?? null,
    content: input.content,
    updatedAt: new Date(),
  }

  if (input.id) {
    const existing = await db.query.blogPosts.findFirst({
      where: eq(schema.blogPosts.id, input.id),
      columns: { publishedAt: true },
    })
    const publishedAt =
      existing?.publishedAt ?? (input.status === "published" ? new Date() : null)
    await db
      .update(schema.blogPosts)
      .set({ ...fields, publishedAt })
      .where(eq(schema.blogPosts.id, input.id))
    return { id: input.id }
  }

  const [row] = await db
    .insert(schema.blogPosts)
    .values({ ...fields, publishedAt: input.status === "published" ? new Date() : null })
    .returning({ id: schema.blogPosts.id })
  return { id: row!.id }
}

export async function deletePost(id: string): Promise<void> {
  await db.delete(schema.blogPosts).where(eq(schema.blogPosts.id, id))
}
```

// One-time migration of the file-based posts into the table. Idempotent: a
// slug that already exists is updated in place, so re-running never duplicates.
// Reads the repo's content/blog (present at dev/build time).
export async function importFilePosts(): Promise<{ imported: number }> {
  const dir = path.join(process.cwd(), "content/blog")
  if (!fs.existsSync(dir)) return { imported: 0 }

  const files = fs.readdirSync(dir).filter((f) => f.endsWith(".md"))
  for (const file of files) {
    const raw = fs.readFileSync(path.join(dir, file), "utf8")
    const { data, content } = matter(raw)
    const slug = (data.slug as string) ?? file.replace(/\.md$/, "")
    // Strip the leading H1; the page renders the title from frontmatter.
    const body = content.replace(/^\s*#\s+.*\r?\n+/, "")
    const existing = await db.query.blogPosts.findFirst({
      where: eq(schema.blogPosts.slug, slug),
      columns: { id: true },
    })
    await savePost({
      id: existing?.id,
      slug,
      title: data.title as string,
      description: data.description as string,
      targetKeyword: (data.targetKeyword as string) ?? undefined,
      secondaryKeywords: (data.secondaryKeywords as string[]) ?? [],
      intent: data.intent as SavePostInput["intent"],
      status: (data.status as SavePostInput["status"]) ?? "draft",
      order: typeof data.order === "number" ? data.order : undefined,
      content: body,
    })
  }

  const [{ n }] = await db.select({ n: sql<number>`count(*)::int` }).from(schema.blogPosts)
  return { imported: Number(n) }
}
```

"use server"

import { revalidatePath } from "next/cache"

import { requireAdmin } from "@/lib/auth"
import { deletePost, importFilePosts, savePost, type SavePostInput } from "@/lib/admin-blog"

function revalidateBlog(slug?: string) {
  revalidatePath("/blog")
  revalidatePath("/admin/blog")
  if (slug) revalidatePath(`/blog/${slug}`)
}

// Returns {error} instead of throwing so the client form can show it inline.
export async function savePostAction(input: SavePostInput): Promise<{ id?: string; error?: string }> {
  await requireAdmin()
  try {
    const { id } = await savePost(input)
    revalidateBlog(input.slug)
    return { id }
  } catch (e) {
    return { error: e instanceof Error ? e.message : "Could not save" }
  }
}

export async function deletePostAction(id: string): Promise<void> {
  await requireAdmin()
  await deletePost(id)
  revalidateBlog()
}

export async function importPostsAction(): Promise<void> {
  await requireAdmin()
  await importFilePosts()
  revalidateBlog()
}
```

```tsx
"use client"

import { useRouter } from "next/navigation"
import { useState, useTransition } from "react"

import { Markdown } from "@/components/blog/markdown"
import type { SavePostInput } from "@/lib/admin-blog"
import { deletePostAction, savePostAction } from "@/app/admin/blog/actions"

type Initial = SavePostInput & { id?: string }

export function BlogEditor({ initial }: { initial: Initial }) {
  const router = useRouter()
  const [pending, start] = useTransition()
  const [error, setError] = useState<string | null>(null)
  const [f, setF] = useState<Initial>(initial)

  function set<K extends keyof Initial>(key: K, value: Initial[K]) {
    setF((prev) => ({ ...prev, [key]: value }))
  }

  function save() {
    setError(null)
    start(async () => {
      const res = await savePostAction({
        ...f,
        secondaryKeywords: f.secondaryKeywords,
      })
      if (res.error) setError(res.error)
      else router.push("/admin/blog")
    })
  }

  function remove() {
    if (!initial.id || !confirm("Delete this post?")) return
    start(async () => {
      await deletePostAction(initial.id!)
      router.push("/admin/blog")
    })
  }

  const input = "border-border bg-background w-full rounded-lg border px-3 py-2 text-sm"

  return (
    <div className="space-y-6">
      <div className="flex items-center justify-between">
        <h1 className="text-2xl font-semibold">{initial.id ? "Edit post" : "New post"}</h1>
        <div className="flex gap-2">
          {initial.id ? (
            <button onClick={remove} disabled={pending} className="border-border rounded-lg border px-4 py-2 text-sm hover:bg-secondary">
              Delete
            </button>
          ) : null}
          <button onClick={save} disabled={pending} className="bg-primary text-primary-foreground rounded-lg px-4 py-2 text-sm font-medium hover:opacity-90 disabled:opacity-50">
            {pending ? "Saving…" : "Save"}
          </button>
        </div>
      </div>

      {error ? <p className="text-sm text-red-400">{error}</p> : null}

      <div className="grid gap-3 sm:grid-cols-2">
        <label className="space-y-1">
          <span className="text-muted-foreground text-xs">Title</span>
          <input className={input} value={f.title} onChange={(e) => set("title", e.target.value)} />
        </label>
        <label className="space-y-1">
          <span className="text-muted-foreground text-xs">Slug</span>
          <input className={input} value={f.slug} onChange={(e) => set("slug", e.target.value)} />
        </label>
        <label className="space-y-1 sm:col-span-2">
          <span className="text-muted-foreground text-xs">Description</span>
          <input className={input} value={f.description} onChange={(e) => set("description", e.target.value)} />
        </label>
        <label className="space-y-1">
          <span className="text-muted-foreground text-xs">Target keyword</span>
          <input className={input} value={f.targetKeyword ?? ""} onChange={(e) => set("targetKeyword", e.target.value || undefined)} />
        </label>
        <label className="space-y-1">
          <span className="text-muted-foreground text-xs">Secondary keywords (comma-separated)</span>
          <input
            className={input}
            value={(f.secondaryKeywords ?? []).join(", ")}
            onChange={(e) => set("secondaryKeywords", e.target.value.split(",").map((s) => s.trim()).filter(Boolean))}
          />
        </label>
        <label className="space-y-1">
          <span className="text-muted-foreground text-xs">Intent</span>
          <select className={input} value={f.intent ?? ""} onChange={(e) => set("intent", (e.target.value || undefined) as Initial["intent"])}>
            <option value="">—</option>
            <option value="informational">informational</option>
            <option value="transactional">transactional</option>
            <option value="commercial">commercial</option>
          </select>
        </label>
        <label className="space-y-1">
          <span className="text-muted-foreground text-xs">Status</span>
          <select className={input} value={f.status} onChange={(e) => set("status", e.target.value as Initial["status"])}>
            <option value="draft">draft</option>
            <option value="published">published</option>
          </select>
        </label>
        <label className="space-y-1">
          <span className="text-muted-foreground text-xs">Order</span>
          <input
            type="number"
            className={input}
            value={f.order ?? ""}
            onChange={(e) => set("order", e.target.value === "" ? undefined : Number(e.target.value))}
          />
        </label>
      </div>

      <div className="grid gap-4 lg:grid-cols-2">
        <div className="space-y-1">
          <span className="text-muted-foreground text-xs">Markdown</span>
          <textarea
            className={`${input} min-h-[420px] font-mono`}
            value={f.content}
            onChange={(e) => set("content", e.target.value)}
          />
        </div>
        <div className="space-y-1">
          <span className="text-muted-foreground text-xs">Preview</span>
          <div className="border-border min-h-[420px] rounded-lg border p-4">
            <Markdown>{f.content || "_Nothing to preview yet._"}</Markdown>
          </div>
        </div>
      </div>
    </div>
  )
}
```

import type { Metadata } from "next"
import { notFound } from "next/navigation"

import { getPostAdmin } from "@/lib/admin-blog"

export const metadata: Metadata = { title: "Edit post · Admin", robots: { index: false } }

        title: row.title,
        description: row.description,

        secondaryKeywords: (row.secondaryKeywords as string[]) ?? [],

        status: row.status,
        order: row.order ?? undefined,
        content: row.content,
      }}
    />
  )
}
